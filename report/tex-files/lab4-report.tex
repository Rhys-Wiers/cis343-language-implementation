        {\Large \textbf{Lab 4: Statements}}
    \end{center}

    \begin{enumerate}
        \item \textbf{Implementation:}
            This lab implements statements in the hashbrown programming language. Currently supported statements are expression statements (which are evaluated and discarded) and print statements (which are evaluated and printed).

        \item \textbf{Test Cases:}
            To run the test cases, navigate to the root directory, run cmake to ensure everything is compiled:
            \begin{minted}{bash}
cmake -B build
cmake --build build
            \end{minted}
                   
            Then run the bash script from the root directory:
            \begin{minted}{bash}
./test/lab4/run-tests.sh
            \end{minted}

            The test cases are listed below, and all passed on my machine:
            \begin{minted}{bash}
> example1.hbn
"one"
true;
print 2 +

> example2.hbn
4 + 2;
print "one";
print true;
print 1 + 2;
print nil;

> test1.hbn
print 10.0 % 3.0 + 2.0 * 4.0;
print "hello" + " " + "world";
print nil;

> test2.hbn
print 0.0 == false;
print 1.0 == true;
print 2.5 == true;
print !("hello" == "world") == (0.0 == false) && (10.0 % 3.0 + 2.0 * 4.0 <= 9.0);

> test3.hbn
10 + 20;
"discarded string expression";
print "only this line should print";
            \end{minted}
    \end{enumerate}
